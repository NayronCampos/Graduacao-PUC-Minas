public static int[] vetorOrdenado(int[] vetA, int[] vetB){
    int M = vetA.length;
    int N = vetB.length;

    int [] vetC = new int[M+N];

    for(int i= (N+M)-1, j=0; i>=0; i--, j++){
        

        if(vetA[i-N]>vetB[i-M]) && ((i-N)>=0) && ((i-M)>=0){
            vetC[j] = vetB[i-M];
        }
        else if ((i-N)>=0) && ((i-M)>=0){
            vetC[j] = vetA[i-N];
        }

        else if((i-N)<=0){
            vetC[j] = vetB[i-M];
        }
        else{
            vetC[j] = vetA[i-N];
        }
    }
}
